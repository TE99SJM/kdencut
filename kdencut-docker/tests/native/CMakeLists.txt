cmake_minimum_required(VERSION 3.16)
project(KdenCutMobileChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test QuickWidgets Xml)
find_package(KF6I18n REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../make_test_catalog.py ${CMAKE_CURRENT_BINARY_DIR}/translations COMMAND_ERROR_IS_FATAL ANY)
add_executable(check_mobile_workspace
    check_mobile_workspace.cpp
    ../../../src/kdencut/mobileworkspace.cpp
    ../../../src/kdencut/mobileworkspace.h)
target_link_libraries(check_mobile_workspace PRIVATE Qt6::Widgets Qt6::Test Qt6::QuickWidgets KF6::I18n)
target_compile_definitions(check_mobile_workspace PRIVATE KDENCUT_TEST_LOCALE="${CMAKE_CURRENT_BINARY_DIR}/translations" KDENCUT_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../..")
find_package(KDDockWidgets-qt6 2.4.0 QUIET)
if(TARGET KDAB::kddockwidgets)
    target_link_libraries(check_mobile_workspace PRIVATE KDAB::kddockwidgets)
    target_compile_definitions(check_mobile_workspace PRIVATE CHECK_KDDOCK)
endif()
enable_testing()
add_executable(check_touch_theme check_touch_theme.cpp)
target_link_libraries(check_touch_theme PRIVATE Qt6::Widgets Qt6::Test)
target_compile_definitions(check_touch_theme PRIVATE KDENCUT_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../..")
add_test(NAME touch_theme COMMAND check_touch_theme)
set_tests_properties(touch_theme PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(check_media_bridge check_media_bridge.cpp ../../../src/kdencut/mediaimportbridge.cpp)
target_link_libraries(check_media_bridge PRIVATE Qt6::Widgets Qt6::Test)
add_test(NAME media_bridge COMMAND check_media_bridge)
set_tests_properties(media_bridge PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_test(NAME mobile_workspace COMMAND check_mobile_workspace)
set_tests_properties(mobile_workspace PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")

add_executable(check_phone_presets check_phone_presets.cpp)
target_link_libraries(check_phone_presets PRIVATE Qt6::Test Qt6::Xml)
target_compile_definitions(check_phone_presets PRIVATE KDENCUT_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../..")
add_test(NAME phone_presets COMMAND check_phone_presets)
