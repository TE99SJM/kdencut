cmake_minimum_required(VERSION 3.16)
project(KdenCutSettingsChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOUIC_SEARCH_PATHS "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/ui")
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
find_package(KF6KIO REQUIRED)
find_package(KF6Completion REQUIRED)
find_package(KF6ItemViews REQUIRED)
find_package(KF6WidgetsAddons REQUIRED)
enable_testing()
foreach(target check_mobile_settings check_project_settings_ui check_touch_theme check_mobile_chrome)
    add_executable(${target} ../native/${target}.cpp
        ../../../src/kdencut/mobilesettings.cpp ../../../src/kdencut/controls.qrc)
    target_link_libraries(${target} PRIVATE Qt6::Widgets Qt6::Test)
    target_compile_definitions(${target} PRIVATE KDENCUT_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../..")
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endforeach()
target_link_libraries(check_project_settings_ui PRIVATE KF6::KIOWidgets KF6::Completion KF6::ItemViews KF6::WidgetsAddons)
