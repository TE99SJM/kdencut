cmake_minimum_required(VERSION 3.16)
project(Osmo360Editor LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Widgets)

add_library(osmo360-media STATIC camera.cpp calibration.cpp media.cpp spherical.cpp)
target_compile_features(osmo360-media PUBLIC cxx_std_17)
target_link_libraries(osmo360-media PUBLIC Qt6::Gui)
qt_add_executable(osmo360-editor editor.cpp install.cpp)
target_link_libraries(osmo360-editor PRIVATE osmo360-media Qt6::Widgets)
if(COMMAND kde_target_enable_exceptions)
  kde_target_enable_exceptions(osmo360-media PUBLIC)
endif()
if(MSVC)
  target_compile_options(osmo360-media PUBLIC /utf-8)
endif()
if(WIN32)
  set_target_properties(osmo360-editor PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
include(GNUInstallDirs)
set(osmo_bin_dir "${CMAKE_INSTALL_BINDIR}")
if(KDE_INSTALL_BINDIR)
  set(osmo_bin_dir "${KDE_INSTALL_BINDIR}")
endif()
if(APPLE AND KDE_INSTALL_BINDIR)
  set(osmo_bin_dir "${KDE_INSTALL_BINDIR}/kdenlive.app/Contents/MacOS")
endif()
set_target_properties(osmo360-editor PROPERTIES MACOSX_BUNDLE FALSE)
install(TARGETS osmo360-editor RUNTIME DESTINATION "${osmo_bin_dir}")
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE-PanoForge" DESTINATION share/osmo360)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  include(CTest)
  if(BUILD_TESTING)
    add_executable(osmo360-media-test tests/media_test.cpp)
    target_link_libraries(osmo360-media-test PRIVATE osmo360-media)
    add_test(NAME native-media COMMAND osmo360-media-test)
  endif()
endif()
